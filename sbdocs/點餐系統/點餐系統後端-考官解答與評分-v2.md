# PART A. 填空題答案速查

## A.1 速查表

| 編號 | 章節 | 位置 | 標準答案 | 評分要點 |
|------|------|------|----------|----------|
| ① | ch6 | `Order.items` | `@OneToMany`（含 `mappedBy="order"`、`cascade=CascadeType.ALL`、`orphanRemoval=true`） | 三個屬性缺一扣半；mappedBy 不可寫成 order_id |
| ② | ch6 | `OrderItem.order` | `@ManyToOne` | 多那一側必用 ManyToOne |
| ③ | ch6 | `OrderItem.menuItem` | `@ManyToOne` | 同上 |
| ④ | ch6 | 概念題 | `BigDecimal` vs `double`：浮點數無法精確表示十進位小數（0.1+0.2≠0.3），金錢運算會累積誤差；BigDecimal 為定點十進位 | 提到「浮點誤差」+「金額不允許誤差」即可給分 |
| ⑤ | ch7 | `searchByName` | `LOWER(m.name) LIKE LOWER(CONCAT('%', :keyword, '%'))` | 大小寫不分＋模糊；只寫 `LIKE '%{:keyword}%'` 可給 4/6 |
| ⑥ | ch7 | `getCategorySalesSummary` | SELECT `m.category, COUNT(m.id), SUM(oi.quantity), SUM(oi.unitPrice * oi.quantity)` | 必須含分組欄位＋3 聚合；少聚合扣分 |
| ⑦ | ch7 | `OrderRepository` | `long countByStatus(String status);` | Derived Query 文法 `countBy<欄位>` |
| ⑧ | ch8 | `MenuItemRequest.name` | `@NotBlank` | null／空字串／全空白三種都擋 |
| ⑨ | ch8 | `MenuItemRequest.price` | `@DecimalMin(value = "0", inclusive = false)` | 價 > 0 |
| ⑩ | ch8 | `CreateOrderRequest.items` | `@Valid` + `@NotEmpty` 兩個疊加 | 缺 @Valid → 巢狀不驗；缺 @NotEmpty → 空集合可過 |
| ⑪ | ch9 | `createOrder` 庫存檢查 | `if (menu.getStockQuantity() < itemReq.getQuantity()) throw new StockNotEnoughException("庫存不足: " + menu.getName() + " (剩 " + menu.getStockQuantity() + ")");` | 判斷錯誤可 4/6；有停售擋但不檢查庫存 → 4 |
| ⑫ | ch9 | `createOrder` 總價累加 | `total = total.add(oi.getSubtotal());` | 用 `add()` 新值；寫 `total += oi.getSubtotal()`（型別錯）→ 2/4 |
| ⑬ | ch9 | `getDashboard` 待處理 | `orderRepo.countByStatus("PENDING")` | 呼叫已定義的 Repository 方法 |
| ⑭ | ch10 | `GET /api/menu/sold-out` | `sold-out` | 需在 `/{id}` 之前宣告 |
| ⑮ | ch10 | `GET /api/orders/dashboard` | `dashboard` | 需在 `/{id}` 之前宣告 |
| ⑯ | ch11 | `StockNotEnoughException` | `public class StockNotEnoughException extends RuntimeException { public StockNotEnoughException(String message) { super(message); } }` | 兩行皆對 |
| ⑰ | ch11 | `GlobalExceptionHandler` | `@ExceptionHandler(StockNotEnoughException.class)` | 屬性值是 class literal |

> 空格標記 `⑯ _________________` 為**兩處**（類別宣告行＋建構子第一行），答案速查涵蓋兩者。

```java
package com.example.order.model;

import jakarta.persistence.*;
import java.math.BigDecimal;

@Entity
@Table(name = "order_items")
public class OrderItem {

    @Id
    @GeneratedValue(strategy = GenerationType.IDENTITY)
    private Long id;

    @ManyToOne
    @JoinColumn(name = "order_id")
    private Order order;

    @ManyToOne
    @JoinColumn(name = "menu_item_id")
    private MenuItem menuItem;

    @Column(name = "quantity", nullable = false)
    private int quantity;

    @Column(name = "unit_price", nullable = false)

    public int getQuantity() { return quantity; }
    public void setQuantity(int quantity) { this.quantity = quantity; }
```

```java
import org.springframework.data.repository.query.Param;

import java.util.List;

public interface MenuItemRepository extends JpaRepository<MenuItem, Long> {

    List<MenuItem> findByCategory(String category);

    List<MenuItem> findByAvailableTrue();

    List<MenuItem> findByStockQuantityAndAvailableTrue(int stockQuantity);

    @Query("SELECT m FROM MenuItem m WHERE " +
           "LOWER(m.name) LIKE LOWER(CONCAT('%', :keyword, '%'))")
    List<MenuItem> searchByName(@Param("keyword") String keyword);

    @Query("SELECT m.name, SUM(oi.quantity) FROM OrderItem oi " +
           "JOIN oi.menuItem m GROUP BY m.id, m.name " +
           "ORDER BY SUM(oi.quantity) DESC")
    List<Object[]> findTopSellingItems();

    @Query("SELECT m.category, COUNT(m.id), SUM(oi.quantity), " +
           "SUM(oi.unitPrice * oi.quantity) FROM OrderItem oi " +
           "JOIN oi.menuItem m GROUP BY m.category " +
           "ORDER BY SUM(oi.quantity) DESC")
```

```java
package com.example.order.service;

import com.example.order.dto.MenuItemRequest;
import com.example.order.exception.ResourceNotFoundException;
import com.example.order.model.MenuItem;
import com.example.order.repository.MenuItemRepository;
import org.springframework.stereotype.Service;
import org.springframework.transaction.annotation.Transactional;

import java.util.List;

@Service
public class MenuItemService {

    private final MenuItemRepository menuRepo;

    public MenuItemService(MenuItemRepository menuRepo) {
        this.menuRepo = menuRepo;
    }

    @Transactional(readOnly = true)
    public List<MenuItem> findAll() { return menuRepo.findAll(); }

    @Transactional(readOnly = true)
    public List<MenuItem> findByCategory(String category) {
        return menuRepo.findByCategory(category);
    }

    @Transactional(readOnly = true)
    public List<MenuItem> findAvailable() {
        return menuRepo.findByAvailableTrue();
    }

    @Transactional(readOnly = true)
    public List<MenuItem> search(String keyword) {
        return menuRepo.searchByName(keyword);
    }

    @Transactional(readOnly = true)
    public List<MenuItem> findSoldOut() {
        return menuRepo.findByStockQuantityAndAvailableTrue(0);
    }

    @Transactional(readOnly = true)
    public MenuItem findById(Long id) {
        return menuRepo.findById(id)
                .orElseThrow(() -> new ResourceNotFoundException("Menu item not found: " + id));
    }

    @Transactional
    public MenuItem create(MenuItemRequest req) {
        MenuItem m = new MenuItem();
        m.setName(req.getName());
        m.setCategory(req.getCategory());
        m.setPrice(req.getPrice());
        m.setAvailable(req.isAvailable());
        m.setStockQuantity(req.getStockQuantity());
        return menuRepo.save(m);
    }

    @Transactional
    public MenuItem update(Long id, MenuItemRequest req) {
        MenuItem m = findById(id);
        m.setName(req.getName());
        m.setCategory(req.getCategory());
        m.setPrice(req.getPrice());
        m.setAvailable(req.isAvailable());
        m.setStockQuantity(req.getStockQuantity());
        return menuRepo.save(m);

    public void delete(Long id) {
        MenuItem m = findById(id);

    }
}
```

## B.14 service/OrderService.java（填空 ⑪⑫⑬ 解答）

```java
package com.example.order.service;

import com.example.order.dto.CreateOrderRequest;
import com.example.order.exception.ResourceNotFoundException;
import com.example.order.exception.StockNotEnoughException;
import com.example.order.model.MenuItem;
import com.example.order.model.Order;
import com.example.order.model.OrderItem;
import com.example.order.repository.MenuItemRepository;
import com.example.order.repository.OrderRepository;
import org.springframework.stereotype.Service;
import org.springframework.transaction.annotation.Transactional;

import java.math.BigDecimal;
import java.time.LocalDate;
import java.time.LocalDateTime;
import java.time.LocalTime;
import java.util.ArrayList;
import java.util.HashMap;
import java.util.List;
import java.util.Map;

@Service
public class OrderService {

    private final OrderRepository    orderRepo;
    private final MenuItemRepository menuRepo;

    public OrderService(OrderRepository orderRepo, MenuItemRepository menuRepo) {
        this.orderRepo = orderRepo;
        this.menuRepo  = menuRepo;
    }

    @Transactional
    public Order createOrder(CreateOrderRequest req) {
        Order order = new Order();
        order.setCustomerName(req.getCustomerName());
        order.setOrderTime(LocalDateTime.now());
        order.setStatus("PENDING");

        BigDecimal total = BigDecimal.ZERO;

        for (CreateOrderRequest.OrderItemRequest itemReq : req.getItems()) {
            MenuItem menu = menuRepo.findById(itemReq.getMenuItemId())
                    .orElseThrow(() -> new ResourceNotFoundException(
                            "Menu item not found: " + itemReq.getMenuItemId()));

            if (!menu.isAvailable()) {
                throw new StockNotEnoughException("已停售: " + menu.getName());
            }
            if (menu.getStockQuantity() < itemReq.getQuantity()) {
                throw new StockNotEnoughException(
                        "庫存不足: " + menu.getName() + " (剩 " + menu.getStockQuantity() + ")");
            }

            menu.setStockQuantity(menu.getStockQuantity() - itemReq.getQuantity());
            menuRepo.save(menu);

            OrderItem oi = new OrderItem();
            oi.setMenuItem(menu);
            oi.setQuantity(itemReq.getQuantity());
            oi.setUnitPrice(menu.getPrice());
            order.addItem(oi);

            total = total.add(oi.getSubtotal());
        }

        order.setTotalAmount(total);
        return orderRepo.save(order);
    }

    @Transactional(readOnly = true)
    public List<Order> findAll() { return orderRepo.findAll(); }

    @Transactional(readOnly = true)
    public Order findById(Long id) {
        return orderRepo.findById(id)
                .orElseThrow(() -> new ResourceNotFoundException("Order not found: " + id));
    }

    @Transactional(readOnly = true)
    public List<Order> findByStatus(String status) {
        return orderRepo.findByStatus(status);
    }

    @Transactional
    public Order updateStatus(Long id, String status) {
        Order o = findById(id);
        o.setStatus(status);
        return orderRepo.save(o);
    }

    @Transactional(readOnly = true)
    public Map<String, Object> getDashboard() {
        LocalDateTime dayStart = LocalDateTime.of(LocalDate.now(), LocalTime.MIN);
        LocalDateTime now      = LocalDateTime.now();

        List<Map<String, Object>> top = new ArrayList<>();
        for (Object[] row : menuRepo.findTopSellingItems()) {
            Map<String, Object> item = new HashMap<>();
            item.put("name", row[0]);
            item.put("totalQuantity", row[1]);
            top.add(item);
        }

        Map<String, Object> result = new HashMap<>();
        result.put("topSelling", top);
        result.put("todayRevenue", orderRepo.sumCompletedAmount(dayStart, now));
        result.put("todayOrders", orderRepo.countByOrderTimeBetween(dayStart, now));
        result.put("pendingOrders", orderRepo.countByStatus("PENDING"));
        return result;
    }
}
```

> 評分重點：`createOrder` 若寫得對，**選錯順序**（先存訂單再扣庫存）不扣分，但**沒有防呆**（停售／庫存）時扣。`getDashboard` 回傳欄位名稱合理即可，Map 或自訂 DTO 皆可。
## B.15 controller/MenuItemController.java（填空 ⑭ 解答）

```java
package com.example.order.controller;

import com.example.order.dto.ApiResponse;
import com.example.order.dto.MenuItemRequest;
import com.example.order.service.MenuItemService;
import jakarta.validation.Valid;
import org.springframework.http.HttpStatus;
import org.springframework.http.ResponseEntity;
import org.springframework.web.bind.annotation.*;

@RestController
@RequestMapping("/api/menu")
public class MenuItemController {

    private final MenuItemService menuService;

    public MenuItemController(MenuItemService menuService) {
        this.menuService = menuService;
    }

    @GetMapping
    public ResponseEntity<ApiResponse<Object>> getAll(
            @RequestParam(required = false) String category,
            @RequestParam(required = false) String keyword,
            @RequestParam(required = false) Boolean available) {
        Object result;
        if (category != null && !category.isBlank()) {
            result = menuService.findByCategory(category);
        } else if (keyword != null && !keyword.isBlank()) {
            result = menuService.search(keyword);
        } else if (available != null && available) {
            result = menuService.findAvailable();
        } else {
            result = menuService.findAll();
        }
        return ResponseEntity.ok(ApiResponse.ok(result));
    }

    // 固定路徑「sold-out」必須宣告在 /{id} 之前
    @GetMapping("/sold-out")
    public ResponseEntity<ApiResponse<Object>> getSoldOut() {
        return ResponseEntity.ok(ApiResponse.ok(menuService.findSoldOut()));
    }

    @GetMapping("/{id}")
```

```java
// GlobalExceptionHandler.java（填空 ⑰）
package com.example.order.exception;

import org.springframework.http.HttpStatus;
import org.springframework.http.ResponseEntity;
import org.springframework.validation.FieldError;
import org.springframework.web.bind.MethodArgumentNotValidException;
import org.springframework.web.bind.annotation.ExceptionHandler;
import org.springframework.web.bind.annotation.RestControllerAdvice;

import java.util.HashMap;
import java.util.Map;

@RestControllerAdvice
public class GlobalExceptionHandler {

    @ExceptionHandler(ResourceNotFoundException.class)
    public ResponseEntity<Map<String, Object>> handleNotFound(ResourceNotFoundException ex) {
        return buildError(HttpStatus.NOT_FOUND, "NOT_FOUND", ex.getMessage());
    }

    @ExceptionHandler(StockNotEnoughException.class)
    public ResponseEntity<Map<String, Object>> handleStock(StockNotEnoughException ex) {
        return buildError(HttpStatus.BAD_REQUEST, "STOCK_NOT_ENOUGH", ex.getMessage());
    }

    @ExceptionHandler(IllegalArgumentException.class)
    public ResponseEntity<Map<String, Object>> handleBadRequest(IllegalArgumentException ex) {
        return buildError(HttpStatus.BAD_REQUEST, "BAD_REQUEST", ex.getMessage());
    }

    @ExceptionHandler(MethodArgumentNotValidException.class)
    public ResponseEntity<Map<String, Object>> handleValidation(MethodArgumentNotValidException ex) {
        Map<String, String> fieldErrors = new HashMap<>();
        for (FieldError fe : ex.getBindingResult().getFieldErrors()) {
            fieldErrors.put(fe.getField(), fe.getDefaultMessage());
        }
        Map<String, Object> body = new HashMap<>();
        body.put("success", false);
        body.put("errorCode", "VALIDATION_ERROR");
        body.put("errors", fieldErrors);
        return ResponseEntity.badRequest().body(body);
    }

    private ResponseEntity<Map<String, Object>> buildError(
            HttpStatus status, String errorCode, String message) {
        Map<String, Object> body = new HashMap<>();
        body.put("success", false);
        body.put("errorCode", errorCode);
        body.put("message", message);
        return ResponseEntity.status(status).body(body);
    }
}
```
